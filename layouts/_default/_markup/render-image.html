{{- /* Image render hook: same scheme allowlist as render-link.html. A disallowed image source (javascript:, data:, anything that is not http(s) or a relative path) renders as its alt text. */ -}} {{- $u := .Destination -}} {{- $scheme := findRE `^[a-zA-Z][a-zA-Z0-9+.-]*:` $u 1 -}} {{- $ok := false -}} {{- if not $scheme -}} {{- $ok = true -}} {{- else -}} {{- $s := lower (index $scheme 0) -}} {{- if in (slice "http:" "https:") $s -}}{{- $ok = true -}}{{- end -}} {{- end -}} {{- if $ok -}} {{- else -}} {{ .PlainText }} {{- end -}}
